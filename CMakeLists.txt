	set(TLS_LIBS OpenSSL::SSL)
	get_filename_component(ssl_dir ${OPENSSL_SSL_LIBRARY} DIRECTORY)
	if(EXISTS ${ssl_dir}/libssl.a AND EXISTS ${ssl_dir}/libcrypto.a)
	    # distro archives may need zstd/zlib/brotli: verify they link alone
	    include(CheckCXXSourceCompiles)
	    set(CMAKE_REQUIRED_INCLUDES ${OPENSSL_INCLUDE_DIR})
	    set(CMAKE_REQUIRED_LIBRARIES ${ssl_dir}/libssl.a
		${ssl_dir}/libcrypto.a ${CMAKE_DL_LIBS} Threads::Threads)
	    check_cxx_source_compiles([[
#include <openssl/ssl.h>
int main() { SSL_CTX_free(SSL_CTX_new(TLS_method())); }
]] OPENSSL_STATIC_LINKS)
	    unset(CMAKE_REQUIRED_INCLUDES)
	    unset(CMAKE_REQUIRED_LIBRARIES)
	    if(OPENSSL_STATIC_LINKS)
		set(TLS_STATIC_LIBS ${ssl_dir}/libssl.a ${ssl_dir}/libcrypto.a)
	    endif()
	endif()
    endif()
endif()

if(CHECK_TSAN)
    # TSAN_OPTIONS=halt_on_error=1 test/echotest -l 200000 -e 50 -p 10
    string(TOUPPER "${CHECK_TSAN}" CHECK_TSAN_UC)
    if(CHECK_TSAN_UC MATCHES "^(ON|TRUE|YES|Y|1)$")
	set(SANITIZE_ARG "thread")
    else()
	set(SANITIZE_ARG "${CHECK_TSAN}")
    endif()
    set(SANITIZE_FLAGS "-fsanitize=${SANITIZE_ARG} -fno-omit-frame-pointer")
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_COMPILER_IS_CLANG)
    if(NOT SANITIZE_FLAGS)
	set(SANITIZE_FLAGS "-U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=3")
    endif()
    set(COMMON_C_FLAGS "${SANITIZE_FLAGS} -ffast-math -march=native -pipe -Wall -Wconversion -Werror -Wextra -Wno-attributes -Wformat=2 -Wno-format-nonliteral -Wformat-security -Wno-format-y2k -Wmultichar -Wno-pragmas -Wredundant-decls -Wsign-conversion -Wstrict-aliasing -Wstrict-overflow=2 -Wno-unknown-pragmas -Wwrite-strings -fstack-protector-strong -fstrict-flex-arrays=2")
    set(COMMON_CXX_FLAGS "-fno-exceptions -fno-rtti -Wno-deprecated -Wno-deprecated-declarations -Wnon-virtual-dtor -Wextra-semi -Woverloaded-virtual -Wsign-promo")
    if(CMAKE_COMPILER_IS_GNUCXX)
	set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fstack-clash-protection")
	if(NOT CHECK_CLANG_TIDY)
	    set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fmax-errors=8 -fno-semantic-interposition -Wdouble-promotion -Wduplicated-branches -Wduplicated-cond -Wformat-overflow=2 -Wformat-signedness -Wformat-truncation=2 -Wlogical-op -Wnull-dereference -Wodr -Wshadow -Wstack-usage=16384 -Wstringop-overflow=3 -Wtrampolines")
	endif()
	set(COMMON_C_FLAGS_RELWITHDEBINFO "-O3 -femit-struct-debug-reduced -ggdb1 -gz")
    else()
	set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fno-standalone-debug -Wdouble-promotion -Wnull-dereference -Wno-null-pointer-subtraction -Wodr -Wshadow -Wno-unknown-warning-option")
	set(COMMON_C_FLAGS_RELWITHDEBINFO "-O3 -gline-tables-only")
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i[3-6]86|x86)$")
	set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fcf-protection=full")
    endif()
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} ${IPO_LDFLAGS} ${SANITIZE_FLAGS}")
    if(APPLE)
	set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,-dead_strip")
	set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS}")
    else()
	if(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
	    # native ld: -z ignore drops unused dependencies and sections
	    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,-z,ignore,-z,now")
	    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS}")
	    set(LD_NO_UNDEFINED "-Wl,-z,defs")
	else()
	    # GOT calls skip PLT stubs; -z now binds the rest at load
	    set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fno-plt")
	    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--as-needed,--exclude-libs,ALL,-gc-sections,-z,now,-z,relro")
	    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,-Bsymbolic-functions")
	    set(LD_NO_UNDEFINED "-Wl,--no-undefined")
	endif()
	if(CMAKE_COMPILER_IS_GNUCXX)	# clang sanitizer runtimes link into exes only
	    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} ${LD_NO_UNDEFINED}")
	endif()
	set(LDFLAGS_C_STATIC "-static-libgcc")
	if(COMPILE_LIBCPP)
	    set(LDFLAGS_CXX_STATIC "${LDFLAGS_C_STATIC} -nostdlib++ -l:libc++.a -l:libc++abi.a")
	else()
	    set(LDFLAGS_CXX_STATIC "${LDFLAGS_C_STATIC} -static-libstdc++")
	endif()
    endif()
    if(PGO STREQUAL "generate")
	set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fprofile-generate=${PGO_DIR} -fprofile-update=atomic")
    elseif(PGO STREQUAL "use")
	if(CMAKE_COMPILER_IS_GNUCXX)
	    set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fprofile-use=${PGO_DIR} -fprofile-partial-training -Wno-missing-profile")
	else()
	    set(COMMON_C_FLAGS "${COMMON_C_FLAGS} -fprofile-use=${PGO_DIR}/default.profdata")
	endif()
    endif()
    set(COMMON_C_FLAGS_DEBUG "-O1 -fno-inline -fno-omit-frame-pointer -fno-optimize-sibling-calls -ggdb -gz -D_DEBUG")
    set(COMMON_C_FLAGS_MINSIZEREL "-Os")
    set(COMMON_C_FLAGS_RELEASE "-fdata-sections -ffunction-sections")
elseif(MSVC)
    set(COMMON_C_FLAGS "/fp:fast /FC /GF /Gy /MP /Qpar /W4 /WX /Zc:inline")
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|x86)$")
	set(COMMON_C_FLAGS "/arch:AVX2 ${COMMON_C_FLAGS}")
    endif()
    set(COMMON_CXX_FLAGS "/EHs-c- /GR-")
    string(REGEX REPLACE "/EHsc|/GR" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
    # CMake defaults supply /Od /RTC1, /O1 and /O2; /Ob0 conflicts (D9025)
    foreach(lang C CXX)
	string(REPLACE "/Ob0" "" CMAKE_${lang}_FLAGS_DEBUG "${CMAKE_${lang}_FLAGS_DEBUG}")
    endforeach()
    # CRT comes from CMAKE_MSVC_RUNTIME_LIBRARY: DLL by default
    set(COMMON_C_FLAGS_DEBUG "/JMC /Ob1 /sdl")
    set(COMMON_C_FLAGS_MINSIZEREL "/Oi")
    # PDB for profilers; /DEBUG otherwise disables /OPT:REF,ICF
    set(COMMON_C_FLAGS_RELEASE "/GL /Gw /Oy- /Zi")
    foreach(t EXE SHARED)
	foreach(c MINSIZEREL RELEASE)
	    set(CMAKE_${t}_LINKER_FLAGS_${c}
		"${CMAKE_${t}_LINKER_FLAGS_${c}} /DEBUG /OPT:REF /OPT:ICF")
	endforeach()
    endforeach()
endif()

foreach(c MINSIZEREL RELWITHDEBINFO)

if(CHECK_CLANG_TIDY)
    if(PROGRAM_CLANG_TIDY)
	execute_process(COMMAND bash -c "dirname $(dirname $(dirname $(clang -print-resource-dir)))" OUTPUT_VARIABLE CLANG_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
	if(APPLE)
	    execute_process(COMMAND bash -c "dirname ${CLANG_DIR}" OUTPUT_VARIABLE osxdir OUTPUT_STRIP_TRAILING_WHITESPACE)
	    set(osxdir "${osxdir}/SDKs/MacOSX.sdk")
	    file(GLOB CLANG_C_INC ${osxdir}/usr/include)
	    file(GLOB CLANG_CXX_INC ${osxdir}/usr/include/c++/*)
	    file(GLOB CLANG_SYS_INC ${osxdir}/System/Library/Frameworks/Kernel.framework/Versions/Current/Headers)
	else()
	    file(GLOB CLANG_C_INC ${CLANG_DIR}/lib64/clang/*/include)
	    set(CLANG_CXX_INC "")
	    set(CLANG_SYS_INC "")
	endif()
	set(CMAKE_C_CLANG_TIDY ${PROGRAM_CLANG_TIDY} --quiet
	    --extra-arg=-isystem --extra-arg=${CLANG_C_INC})
	set(CMAKE_CXX_CLANG_TIDY ${PROGRAM_CLANG_TIDY} --quiet
	    --extra-arg=-isystem --extra-arg=${CLANG_C_INC}
	    --extra-arg=-std=c++${CMAKE_CXX_STANDARD})
	if(CLANG_SYS_INC)
	    list(APPEND CMAKE_C_CLANG_TIDY --extra-arg=-isystem --extra-arg=${CLANG_SYS_INC})
	endif()
	if(CLANG_CXX_INC)
	    list(APPEND CMAKE_CXX_CLANG_TIDY --extra-arg=-isystem --extra-arg=${CLANG_CXX_INC})
	endif()
	if(CLANG_SYS_INC)
	    list(APPEND CMAKE_CXX_CLANG_TIDY --extra-arg=-isystem --extra-arg=${CLANG_SYS_INC})
	endif()
	if(GCC_INSTALL_ARG)
	    list(APPEND CMAKE_C_CLANG_TIDY --extra-arg=${GCC_INSTALL_ARG})
	    list(APPEND CMAKE_CXX_CLANG_TIDY --extra-arg=${GCC_INSTALL_ARG})
	endif()
	set(CMAKE_DISABLE_PRECOMPILE_HEADERS TRUE)
    else()
	message(AUTHOR_WARNING "clang-tidy not available")
    endif()
